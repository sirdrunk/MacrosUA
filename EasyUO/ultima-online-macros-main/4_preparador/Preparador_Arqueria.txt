; ################################################################
; # PREPARADOR - ARQUERÍA CASTILLO
; ################################################################
;
; Control de movimiento entre puntos del castillo:
; - gestión de HP (curación)
; - alternancia INICIAL / FINAL
; - limpieza de vendas
;


set %limitHits 15


; ============================================================
; FLUJO PRINCIPAL
; ============================================================

Preparador:

    gosub ControlMovimiento
    gosub LimpiarVendas
    gosub ControlMovimiento

exit


; ============================================================
; CONTROL DE MOVIMIENTO
; ============================================================

sub ControlMovimiento

    ; prioridad 1: curación
    if #HITS < %limitHits
    {
        Event SysMessage [MOVE] zona curación
        gosub MoverA *CFG_HEAL_POS_X *CFG_HEAL_POS_Y
        return
    }

    ; prioridad 2: ciclo entre puntos
    gosub EstoyCerca *CFG_INITIAL_POS_X *CFG_INITIAL_POS_Y

    if %estoyCerca = 1
    {
        Event SysMessage [MOVE] hacia FINAL
        gosub MoverA *CFG_FINAL_POS_X *CFG_FINAL_POS_Y
        return
    }

    gosub EstoyCerca *CFG_FINAL_POS_X *CFG_FINAL_POS_Y

    if %estoyCerca = 1
    {
        Event SysMessage [MOVE] hacia INICIAL
        gosub MoverA *CFG_INITIAL_POS_X *CFG_INITIAL_POS_Y
        return
    }

    ; fallback
    Event SysMessage [MOVE] recolocación INICIAL
    gosub MoverA *CFG_INITIAL_POS_X *CFG_INITIAL_POS_Y

return


; ============================================================
; CHECK PROXIMIDAD
; ============================================================

sub EstoyCerca

    if #CHARPOSX >= %1 - 1 && #CHARPOSX <= %1 + 1 && #CHARPOSY >= %2 - 1 && #CHARPOSY <= %2 + 1
        set %estoyCerca 1
    else
        set %estoyCerca 0

return


; ============================================================
; MOVIMIENTO
; ============================================================

sub MoverA

    if #CHARPOSX = %1 && #CHARPOSY = %2
        return

    move %1 %2 0 20s

return


; ============================================================
; LIMPIEZA DE VENDAS
; ============================================================

sub LimpiarVendas

    set %typeVendasSucias AMF

    finditem %typeVendasSucias C_ , #BACKPACKID

    if #FINDCNT > 0
    {
        gosub SoltarPilon
        gosub UsarVendasSucias
        gosub RecogerPilon
    }

    wait 5s

return


; ============================================================
; SOLTAR PILÓN
; ============================================================

sub SoltarPilon

    set %idPilonAgua *CFG_PILON_AGUA_ID

    set %dropX #CHARPOSX + 1
    set %dropY #CHARPOSY + 1

    Exevent Drag %idPilonAgua
    wait 10

    Exevent Dropg #CHARPOSX #CHARPOSY #CHARPOSZ
    wait 10

return


; ============================================================
; USAR VENDAS
; ============================================================

sub UsarVendasSucias

    set #LOBJECTID #FINDID
    event macro 17 0

    target 10s
	set #LTARGETID %idPilonAgua
	set #LTARGETKIND 1

    event macro 22 0

    wait 1s

return


; ============================================================
; RECOGER PILÓN
; ============================================================

sub RecogerPilon

    Exevent Drag %idPilonAgua
    wait 10

    Exevent Dropc #BACKPACKID
    wait 10

return
