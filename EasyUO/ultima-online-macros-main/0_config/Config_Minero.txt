Event SysMessage [CONFIG] Inicializando configuracion de MINA


; =========================
; INSTRUCCIONES
; =========================


; Configurar previamente la farmzone con el CONFIG_GENERATOR_FARMZONE
; Configurar previamente la casa con el CONFIG_GENERATOR_HOME
; Una farmzone es una extension cuadrada donde el personaje intentará minar.
; *IMPORTANTE, debe poderse andar sin grandes obstaculos entre los centros de farmzones adyacentes para que no se quede bloqueado el personaje.
; Situarse cerca de la primera farmzone
; La primera farmzone debe estar cerca de la casa.

; =========================
; FICHEROS
; =========================
set *CFG_SCRIPT_ORQUESTADOR Orquestador.txt
set *CFG_SCRIPT_INICIALIZADOR Inicializador_TileZones.txt
set *CFG_SCRIPT_PREPARADOR Preparador_TileFarmer.txt
set *CFG_SCRIPT_EJECUTOR Ejecutor_Skill_On_Tile.txt

; =========================
; MOVIMIENTO
; =========================
set *CFG_IR_CASA 1
set *CFG_TILE_Z 1
set *CFG_TILE_USE_Z 0 - 90

set %SCRIPT_FARMZONE *SCRIPT_DIR , 0_config\farmzone\ , CONFIG_FARMZONE_Mina.txt  ;***********	Farmone a utilizar
call %SCRIPT_FARMZONE

set %SCRIPT_CASA *SCRIPT_DIR , 0_config\home\ , CONFIG_HOME_Mina.txt  ;***********	Casa a utilizar
call %SCRIPT_CASA

; =========================
; TILES OK y BAD
; =========================
set *CFG_TILES_OK0 ROCK
set *CFG_TILES_OK_COUNT 0

set *CFG_TILES_BAD0 PATATA
set *CFG_TILES_BAD_COUNT 0
; =========================
; SKILL
; =========================
set *CFG_SKILL_NAME MINERO
set *CFG_SKILL_USE Mineria
set *CFG_SKILL_REQUIRES_OBJECT 1
set *CFG_SKILL_OBJECT_TYPE NPF      ; se llenará automáticamente
set *CFG_SKILL_MAX_USOS 5
set *CFG_SKILL_OBJECT_EQUIPABLE 1

; =========================
; JOURNAL OK
; =========================
set *CFG_OK0 PONES_EL_MINERAL
set *CFG_OK1 A_TUS_PIES
set *CFG_OK2 YOUR_SKILL_IN
set *CFG_OK_COUNT 2

; =========================
; JOURNAL FAIL
; =========================
set *CFG_FAIL0 AQUI_NO_HAY_NADA
set *CFG_FAIL1 NO_LO_ALCANZAS
set *CFG_FAIL2 PARECE_QUE_ES_INMUNE
set *CFG_FAIL3 YOU_CANNOT_CHOP_SO
set *CFG_FAIL4 ESO_ESTA_MUY_LEJOS
set *CFG_FAIL5 NO_PUEDES_VER
set *CFG_FAIL_COUNT 5

; =========================
; TIEMPOS
; =========================
set *CFG_TIME_ACCION 2s
set *CFG_TIMEOUT_ACCION 120000

; =========================
; ITEMS Y PESO
; =========================
set *CFG_ITEM_RECOGIDA0 DWJ
set *CFG_ITEM_RECOGIDA1 TVJ
set *CFG_ITEM_RECOGIDA2 GWJ
set *CFG_ITEM_RECOGIDA_COUNT 2
set *CFG_PESO_MAX #MAXWEIGHT - 80


; =========================
; LOG
; =========================
set *CFG_LOG_LEVEL 3

Event SysMessage [CONFIG] MINA lista
exit
