; ============================================================
; EJECUTOR ARQUERIA
; ============================================================
;
; Ejecuta la acción principal de la skill de arquearía.
;
; Responsabilidad:
;   - Ejecutar acción repetida configurada
;   - Controlar número de usos por ciclo
;
; No decide targets.
; No mueve al personaje.
; Solo ejecuta acción definida en config.
;
; ============================================================


Ejecutor:

    set %actionUses 0

    while %actionUses < *CFG_SKILL_ACTION_USES
    {
        set %actionUses %actionUses + 1
        gosub EjecutarAccion
    }

exit

sub EjecutarAccion

    ; ========================================================
